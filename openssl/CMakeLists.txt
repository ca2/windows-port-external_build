cmake_minimum_required(VERSION 3.24)
project(openssl_external LANGUAGES C)
include(ExternalProject)
include(ProcessorCount)

if(NOT WIN32 OR CMAKE_CONFIGURATION_TYPES)
    message(FATAL_ERROR "Use Windows with a single-configuration generator (Ninja) and separate Debug/Release directories")
endif()
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "OpenSSL configuration" FORCE)
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release|RelWithDebInfo|MinSizeRel)$")
    message(FATAL_ERROR "Unsupported CMAKE_BUILD_TYPE: ${CMAKE_BUILD_TYPE}")
endif()
option(__REACTOS__ "Target Windows XP/ReactOS" OFF)
if(__REACTOS__)
    if(NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "XP requires 32-bit MinGW targeting MSVCRT, without UCRT")
    endif()
    include(CheckCSourceCompiles)
    check_c_source_compiles("#include <_mingw.h>\n#ifdef _UCRT\n#error UCRT is unavailable on XP\n#endif\nint main(void) { return 0; }" OPENSSL_XP_MSVCRT_TOOLCHAIN)
    if(NOT OPENSSL_XP_MSVCRT_TOOLCHAIN)
        message(FATAL_ERROR "The XP toolchain must use MSVCRT, not UCRT")
    endif()
    set(default_version 3.0.22)
    set(profile xp)
else()
    set(default_version 3.5.8)
    set(profile windows)
endif()
set(OPENSSL_VERSION "${default_version}" CACHE STRING "Pinned OpenSSL 3 release")
if(NOT OPENSSL_VERSION MATCHES "^3\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "This wrapper supports OpenSSL 3 release archives")
endif()
if(__REACTOS__ AND NOT OPENSSL_VERSION MATCHES "^3\\.0\\.")
    message(FATAL_ERROR "OpenSSL 3.1 and newer dropped XP support; use a 3.0.x release")
endif()
set(OPENSSL_SHA256 "" CACHE STRING "SHA256 for a custom release")
set(OPENSSL_URL "" CACHE STRING "Optional archive URL or local file")
set(OPENSSL_SOURCE_DIR "" CACHE PATH "Optional existing unmodified sources")
set(OPENSSL_INSTALL_DIR "${CMAKE_CURRENT_BINARY_DIR}/install" CACHE PATH "Installed OpenSSL SDK")
set(OPENSSL_EXTRA_CONFIGURE_ARGS "" CACHE STRING "Additional Configure arguments as a CMake list")
foreach(argument IN LISTS OPENSSL_EXTRA_CONFIGURE_ARGS)
    if(argument MATCHES "^(no-shared|no-threads|no-apps|--prefix(=|$)|--libdir(=|$)|--openssldir(=|$)|--cross-compile-prefix(=|$))")
        message(FATAL_ERROR "OPENSSL_EXTRA_CONFIGURE_ARGS cannot override the SDK contract: ${argument}")
    endif()
endforeach()
option(OPENSSL_ASM "Enable upstream assembly optimizations (MSVC needs NASM)" OFF)
option(OPENSSL_RUN_TESTS "Build and run the upstream test suite on the build host" OFF)
option(OPENSSL_VERIFY "Run installed CLI smoke tests on the build host" ON)
ProcessorCount(default_jobs)
if(NOT default_jobs)
    set(default_jobs 4)
endif()
set(OPENSSL_JOBS "${default_jobs}" CACHE STRING "GNU make parallel jobs; NMake is serial")
if(NOT OPENSSL_JOBS MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "OPENSSL_JOBS must be positive")
endif()
get_filename_component(OPENSSL_INSTALL_DIR "${OPENSSL_INSTALL_DIR}" ABSOLUTE)
if(MSVC)
    find_program(OPENSSL_PERL NAMES perl.exe perl HINTS "C:/Strawberry/perl/bin")
    if(NOT OPENSSL_PERL)
        message(FATAL_ERROR "MISSING_DEPENDENCY:perl\nCould not find native Windows Perl (perl.exe). See dependencies/windows/index.html#perl")
    endif()
    execute_process(COMMAND "${OPENSSL_PERL}" -e "print $^O" OUTPUT_VARIABLE perl_os RESULT_VARIABLE perl_os_result)
    if(NOT perl_os_result EQUAL 0 OR NOT perl_os STREQUAL "MSWin32")
        message(FATAL_ERROR "MISSING_DEPENDENCY:perl\nMSVC needs native Windows Perl (Strawberry Perl), not '${perl_os}'. See dependencies/windows/index.html#perl")
    endif()
    find_program(OPENSSL_MAKE NAMES nmake.exe nmake)
    if(NOT OPENSSL_MAKE)
        message(FATAL_ERROR "MISSING_DEPENDENCY:nmake\nCould not find nmake.exe. See dependencies/windows/index.html#nmake")
    endif()
    if(OPENSSL_ASM)
        find_program(OPENSSL_NASM NAMES nasm.exe nasm)
        if(NOT OPENSSL_NASM)
            message(FATAL_ERROR "MISSING_DEPENDENCY:nasm\nOPENSSL_ASM=ON requires nasm.exe. See dependencies/windows/index.html#nasm")
        endif()
    endif()
elseif(MINGW)
    # OpenSSL's MinGW makefiles need MSYS Perl and GNU make, not Strawberry Perl.
    find_program(OPENSSL_BASH NAMES bash.exe bash HINTS "C:/msys64/usr/bin")
    if(NOT OPENSSL_BASH)
        message(FATAL_ERROR "MISSING_DEPENDENCY:msys2\nMinGW OpenSSL build needs MSYS2 bash.exe. See dependencies/windows/index.html#msys2")
    endif()
else()
    message(FATAL_ERROR "Supported Windows compilers: MSVC and MinGW")
endif()
set(hash_3.0.22 67ebca7e50d17383028045486653492195b83db95f8558709701bb47b5c1ef81)
set(hash_3.5.8 a8f84a39918ec6415ce765d9b429d313ba97b8143169c172e734b9514464f5b2)
if(OPENSSL_SHA256)
    set(archive_hash "${OPENSSL_SHA256}")
else()
    set(archive_hash "${hash_${OPENSSL_VERSION}}")
endif()
if(NOT archive_hash AND NOT OPENSSL_SOURCE_DIR)
    message(FATAL_ERROR "Supply OPENSSL_SHA256 for unlisted releases")
endif()
set(build_root "${CMAKE_CURRENT_BINARY_DIR}/openssl-${OPENSSL_VERSION}-${profile}-${CMAKE_SIZEOF_VOID_P}-${CMAKE_BUILD_TYPE}")
set(settings "${CMAKE_CURRENT_BINARY_DIR}/openssl-settings.cmake")
set(settings_content "# Generated OpenSSL settings\n")
foreach(name IN ITEMS CMAKE_C_COMPILER CMAKE_AR CMAKE_RANLIB CMAKE_RC_COMPILER CMAKE_OBJDUMP
        CMAKE_SIZEOF_VOID_P CMAKE_C_COMPILER_ARCHITECTURE_ID CMAKE_BUILD_TYPE MSVC __REACTOS__ OPENSSL_VERSION OPENSSL_BASH
        OPENSSL_PERL OPENSSL_MAKE OPENSSL_NASM OPENSSL_ASM OPENSSL_INSTALL_DIR OPENSSL_JOBS
        OPENSSL_RUN_TESTS OPENSSL_VERIFY OPENSSL_EXTRA_CONFIGURE_ARGS)
    string(APPEND settings_content "set(${name} [==[${${name}}]==])\n")
endforeach()
file(CONFIGURE OUTPUT "${settings}" CONTENT "${settings_content}" @ONLY)
if(OPENSSL_SOURCE_DIR)
    if(NOT EXISTS "${OPENSSL_SOURCE_DIR}/Configure" OR NOT EXISTS "${OPENSSL_SOURCE_DIR}/VERSION.dat")
        message(FATAL_ERROR "OPENSSL_SOURCE_DIR must contain OpenSSL 3 sources")
    endif()
    set(source_arguments SOURCE_DIR "${OPENSSL_SOURCE_DIR}" DOWNLOAD_COMMAND "")
else()
    if(OPENSSL_URL)
        set(download_url "${OPENSSL_URL}")
    else()
        set(download_url "https://github.com/openssl/openssl/releases/download/openssl-${OPENSSL_VERSION}/openssl-${OPENSSL_VERSION}.tar.gz")
    endif()
    set(source_arguments SOURCE_DIR "${build_root}/source" DOWNLOAD_DIR "${CMAKE_CURRENT_BINARY_DIR}/downloads"
        URL "${download_url}" URL_HASH "SHA256=${archive_hash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endif()
if(MSVC AND CMAKE_C_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64")
    set(dll_suffix -arm64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(dll_suffix -x64)
else()
    set(dll_suffix "")
endif()
file(MAKE_DIRECTORY "${OPENSSL_INSTALL_DIR}/include")
set(byproducts "")
foreach(component IN ITEMS crypto ssl)
    if(MSVC)
        set(implib "${OPENSSL_INSTALL_DIR}/lib/lib${component}.lib")
    else()
        set(implib "${OPENSSL_INSTALL_DIR}/lib/lib${component}.dll.a")
    endif()
    set(dll "${OPENSSL_INSTALL_DIR}/bin/lib${component}-3${dll_suffix}.dll")
    list(APPEND byproducts "${implib}" "${dll}")
    add_library(lib${component} SHARED IMPORTED GLOBAL)
    set_target_properties(lib${component} PROPERTIES IMPORTED_IMPLIB "${implib}" IMPORTED_LOCATION "${dll}"
        INTERFACE_INCLUDE_DIRECTORIES "${OPENSSL_INSTALL_DIR}/include" IMPORTED_NO_SYSTEM TRUE)
endforeach()
# Standard FindOpenSSL target spelling, in addition to ca2's libcrypto/libssl names.
add_library(OpenSSL::Crypto ALIAS libcrypto)
add_library(OpenSSL::SSL ALIAS libssl)
set_property(TARGET libssl PROPERTY INTERFACE_LINK_LIBRARIES libcrypto)
ExternalProject_Add(openssl_build ${source_arguments}
    PREFIX "${build_root}" BINARY_DIR "${build_root}/build"
    CONFIGURE_COMMAND "${CMAKE_COMMAND}" "-DSETTINGS_FILE=${settings}"
        "-DSOURCE_DIR=<SOURCE_DIR>" "-DBINARY_DIR=<BINARY_DIR>" -DPHASE=configure -P "${CMAKE_CURRENT_SOURCE_DIR}/configure.cmake"
    BUILD_COMMAND "${CMAKE_COMMAND}" "-DSETTINGS_FILE=${settings}"
        "-DBINARY_DIR=<BINARY_DIR>" -DPHASE=build -P "${CMAKE_CURRENT_SOURCE_DIR}/configure.cmake"
    INSTALL_COMMAND "${CMAKE_COMMAND}" "-DSETTINGS_FILE=${settings}"
        "-DBINARY_DIR=<BINARY_DIR>" -DPHASE=install -P "${CMAKE_CURRENT_SOURCE_DIR}/configure.cmake"
    BUILD_BYPRODUCTS ${byproducts}
    LOG_DOWNLOAD ON LOG_CONFIGURE ON LOG_BUILD ON LOG_INSTALL ON LOG_OUTPUT_ON_FAILURE ON)
ExternalProject_Add_StepDependencies(openssl_build configure "${settings}"
    "${CMAKE_CURRENT_SOURCE_DIR}/configure.cmake" "${CMAKE_CURRENT_SOURCE_DIR}/verify.cmake")
add_dependencies(libcrypto openssl_build)
add_dependencies(libssl openssl_build)
message(STATUS "OpenSSL ${OPENSSL_VERSION}: ${profile}, ${CMAKE_BUILD_TYPE}, install=${OPENSSL_INSTALL_DIR}")

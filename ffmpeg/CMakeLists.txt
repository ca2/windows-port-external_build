cmake_minimum_required(VERSION 3.24)
project(ffmpeg_external LANGUAGES C)
include(ExternalProject)
include(ProcessorCount)
if(CMAKE_CONFIGURATION_TYPES)
    message(FATAL_ERROR "Use a single-configuration generator such as Ninja and separate Debug/Release build directories")
endif()
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "FFmpeg configuration" FORCE)
endif()
set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Debug Release RelWithDebInfo MinSizeRel)
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release|RelWithDebInfo|MinSizeRel)$")
    message(FATAL_ERROR "Unsupported CMAKE_BUILD_TYPE: ${CMAKE_BUILD_TYPE}")
endif()
if(NOT WIN32)
    message(FATAL_ERROR "This wrapper targets Windows with MinGW or MSVC")
endif()
option(__REACTOS__ "Target Windows XP/ReactOS" OFF)
if(__REACTOS__)
    if(NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "XP requires 32-bit MinGW, without UCRT")
    endif()
    set(default_version 6.1.6)
    set(profile xp)
else()
    set(default_version 9.0.2)
    set(profile windows)
endif()
set(FFMPEG_VERSION "${default_version}" CACHE STRING "Pinned FFmpeg release")
set(FFMPEG_SHA256 "" CACHE STRING "SHA256 override for a custom release")
set(FFMPEG_URL "" CACHE STRING "Archive URL or local file; empty uses the official release")
set(FFMPEG_SOURCE_DIR "" CACHE PATH "Optional existing unmodified FFmpeg sources")
set(FFMPEG_INSTALL_DIR "${CMAKE_CURRENT_BINARY_DIR}/install" CACHE PATH "Installed FFmpeg SDK")
set(FFMPEG_EXTRA_CONFIGURE_ARGS "" CACHE STRING "Additional configure arguments as a CMake list")
option(FFMPEG_BUILD_PROGRAMS "Build ffmpeg and ffprobe, but not ffplay" OFF)
option(FFMPEG_NETWORK "Enable network protocols" OFF)
ProcessorCount(default_jobs)
if(NOT default_jobs)
    set(default_jobs 4)
endif()
set(FFMPEG_JOBS "${default_jobs}" CACHE STRING "GNU make parallel jobs")
if(NOT FFMPEG_JOBS MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "FFMPEG_JOBS must be positive")
endif()
find_program(FFMPEG_BASH NAMES bash.exe bash HINTS "C:/msys64/usr/bin")
if(NOT FFMPEG_BASH)
    message(FATAL_ERROR "MISSING_DEPENDENCY:msys2\nCould not find MSYS2 bash.exe. See dependencies/windows/ (x64.html / arm64.html)#msys2")
endif()
get_filename_component(ffmpeg_msys_bin "${FFMPEG_BASH}" DIRECTORY)
if(NOT EXISTS "${ffmpeg_msys_bin}/cygpath.exe")
    message(FATAL_ERROR "MISSING_DEPENDENCY:msys2\nFound bash at '${FFMPEG_BASH}', but cygpath.exe is missing (MSYS2 is required). See dependencies/windows/ (x64.html / arm64.html)#msys2")
endif()
if(NOT EXISTS "${ffmpeg_msys_bin}/make.exe")
    message(FATAL_ERROR "MISSING_DEPENDENCY:msys2\nMSYS2 GNU make is missing (${ffmpeg_msys_bin}/make.exe). Install with 'pacman -S make diffutils'. See dependencies/windows/ (x64.html / arm64.html)#msys2")
endif()
if(NOT EXISTS "${ffmpeg_msys_bin}/diff.exe")
    message(FATAL_ERROR "MISSING_DEPENDENCY:msys2\nMSYS2 diffutils is missing (${ffmpeg_msys_bin}/diff.exe). Install with 'pacman -S diffutils'. See dependencies/windows/ (x64.html / arm64.html)#msys2")
endif()
if(MSVC AND CMAKE_C_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64")
    list(FIND FFMPEG_EXTRA_CONFIGURE_ARGS "--disable-asm" ffmpeg_disable_asm_index)
    if(ffmpeg_disable_asm_index EQUAL -1)
        find_program(FFMPEG_GAS_PREPROCESSOR NAMES gas-preprocessor.pl HINTS "${ffmpeg_msys_bin}" "C:/msys64/usr/local/bin")
        if(NOT FFMPEG_GAS_PREPROCESSOR OR NOT EXISTS "${ffmpeg_msys_bin}/perl.exe")
            message(FATAL_ERROR "MISSING_DEPENDENCY:gas-preprocessor\nFFmpeg ARM64 MSVC build requires gas-preprocessor.pl and MSYS2 perl. See dependencies/windows/arm64.html#gas-preprocessor")
        endif()
    endif()
elseif(MSVC AND NOT CMAKE_C_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64")
    list(FIND FFMPEG_EXTRA_CONFIGURE_ARGS "--disable-x86asm" ffmpeg_disable_x86asm_index)
    if(ffmpeg_disable_x86asm_index EQUAL -1)
        find_program(FFMPEG_X86ASM NAMES nasm.exe nasm yasm.exe yasm HINTS "${ffmpeg_msys_bin}" "C:/Program Files/NASM")
        if(NOT FFMPEG_X86ASM)
            message(FATAL_ERROR "MISSING_DEPENDENCY:nasm\nFFmpeg x86/x64 MSVC build requires nasm (or yasm). See dependencies/windows/x64.html#nasm")
        endif()
    endif()
endif()
set(hash_6.1.2 3b624649725ecdc565c903ca6643d41f33bd49239922e45c9b1442c63dca4e38)
set(hash_6.1.4 a231e3d5742c44b1cdaebfb98ad7b6200d12763e0b6db9e1e2c5891f2c083a18)
set(hash_6.1.6 d4fcb164028dd3beee5d92c0ac72e46aac6973c75ea12dc14de07bf8f407370a)
set(hash_9.0.2 8c3850283eb25fa026482078a04051e0be17347b09ef81a0849bec15a96e002e)
if(FFMPEG_SHA256)
    set(archive_hash "${FFMPEG_SHA256}")
else()
    set(archive_hash "${hash_${FFMPEG_VERSION}}")
endif()
if(NOT archive_hash AND NOT FFMPEG_SOURCE_DIR)
    message(FATAL_ERROR "Supply FFMPEG_SHA256 for unlisted releases")
endif()
if(FFMPEG_VERSION MATCHES "^6\\.1\\.")
    set(component_majors 58 4 60 60 7 9 60)
elseif(FFMPEG_VERSION MATCHES "^9\\.0\\.")
    set(component_majors 61 7 63 63 10 12 63)
else()
    message(FATAL_ERROR "Supported families: 6.1.x and 9.0.x")
endif()
if(__REACTOS__ AND NOT FFMPEG_VERSION MATCHES "^6\\.1\\.")
    message(FATAL_ERROR "Only 6.1.x has XP adapters in this project")
endif()
set(build_root "${CMAKE_CURRENT_BINARY_DIR}/ffmpeg-${FFMPEG_VERSION}-${profile}-${CMAKE_BUILD_TYPE}")
set(settings "${CMAKE_CURRENT_BINARY_DIR}/ffmpeg-settings.cmake")
set(settings_content "# Generated settings\n")
foreach(name IN ITEMS CMAKE_C_COMPILER CMAKE_AR CMAKE_RANLIB CMAKE_NM CMAKE_STRIP
        CMAKE_RC_COMPILER CMAKE_SIZEOF_VOID_P CMAKE_C_COMPILER_ARCHITECTURE_ID CMAKE_BUILD_TYPE MSVC __REACTOS__ FFMPEG_BASH
        FFMPEG_INSTALL_DIR FFMPEG_JOBS FFMPEG_BUILD_PROGRAMS FFMPEG_NETWORK FFMPEG_EXTRA_CONFIGURE_ARGS)
    string(APPEND settings_content "set(${name} [==[${${name}}]==])\n")
endforeach()
file(CONFIGURE OUTPUT "${settings}" CONTENT "${settings_content}" @ONLY)
if(FFMPEG_SOURCE_DIR)
    if(NOT EXISTS "${FFMPEG_SOURCE_DIR}/configure")
        message(FATAL_ERROR "FFMPEG_SOURCE_DIR must contain configure")
    endif()
    if(EXISTS "${FFMPEG_SOURCE_DIR}/RELEASE")
        file(READ "${FFMPEG_SOURCE_DIR}/RELEASE" source_release)
        string(STRIP "${source_release}" source_release)
        if(NOT source_release STREQUAL FFMPEG_VERSION)
            message(FATAL_ERROR "Source release ${source_release} differs from FFMPEG_VERSION=${FFMPEG_VERSION}")
        endif()
    endif()
    set(source_arguments SOURCE_DIR "${FFMPEG_SOURCE_DIR}" DOWNLOAD_COMMAND "")
else()
    if(FFMPEG_URL)
        set(download_url "${FFMPEG_URL}")
    else()
        set(download_url "https://ffmpeg.org/releases/ffmpeg-${FFMPEG_VERSION}.tar.xz")
    endif()
    set(source_arguments SOURCE_DIR "${build_root}/source"
        DOWNLOAD_DIR "${CMAKE_CURRENT_BINARY_DIR}/downloads"
        URL "${download_url}" URL_HASH "SHA256=${archive_hash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endif()
set(components avutil swresample avformat avcodec swscale)
if(FFMPEG_BUILD_PROGRAMS)
    list(APPEND components avfilter avdevice)
endif()
file(MAKE_DIRECTORY "${FFMPEG_INSTALL_DIR}/include")
set(byproducts "")
set(index 0)
foreach(component IN LISTS components)
    list(GET component_majors ${index} major)
    if(MSVC)
        set(implib "${FFMPEG_INSTALL_DIR}/bin/${component}.lib")
    else()
        set(implib "${FFMPEG_INSTALL_DIR}/lib/lib${component}.dll.a")
    endif()
    set(dll "${FFMPEG_INSTALL_DIR}/bin/${component}-${major}.dll")
    list(APPEND byproducts "${implib}" "${dll}")
    add_library(${component} SHARED IMPORTED GLOBAL)
    add_library(FFmpeg::${component} ALIAS ${component})
    set_target_properties(${component} PROPERTIES IMPORTED_IMPLIB "${implib}"
        IMPORTED_LOCATION "${dll}" INTERFACE_INCLUDE_DIRECTORIES "${FFMPEG_INSTALL_DIR}/include")
    math(EXPR index "${index}+1")
endforeach()
ExternalProject_Add(ffmpeg_build ${source_arguments}
    PREFIX "${build_root}" BINARY_DIR "${build_root}/build"
    CONFIGURE_COMMAND "${CMAKE_COMMAND}" "-DSETTINGS_FILE=${settings}"
        "-DSOURCE_DIR=<SOURCE_DIR>" "-DBINARY_DIR=<BINARY_DIR>" -DPHASE=configure
        -P "${CMAKE_CURRENT_SOURCE_DIR}/configure.cmake"
    BUILD_COMMAND "${CMAKE_COMMAND}" "-DSETTINGS_FILE=${settings}"
        "-DBINARY_DIR=<BINARY_DIR>" -DPHASE=build -P "${CMAKE_CURRENT_SOURCE_DIR}/configure.cmake"
    INSTALL_COMMAND "${CMAKE_COMMAND}" "-DSETTINGS_FILE=${settings}"
        "-DBINARY_DIR=<BINARY_DIR>" -DPHASE=install -P "${CMAKE_CURRENT_SOURCE_DIR}/configure.cmake"
    BUILD_BYPRODUCTS ${byproducts}
    LOG_DOWNLOAD ON LOG_CONFIGURE ON LOG_BUILD ON LOG_INSTALL ON LOG_OUTPUT_ON_FAILURE ON)
ExternalProject_Add_StepDependencies(ffmpeg_build configure "${settings}"
    "${CMAKE_CURRENT_SOURCE_DIR}/configure.cmake" "${CMAKE_CURRENT_SOURCE_DIR}/ffmpeg-xp-compat.h")
include("${CMAKE_CURRENT_LIST_DIR}/../stage-headers.cmake")
external_build_stage_headers(ffmpeg_build "${FFMPEG_INSTALL_DIR}")
foreach(component IN LISTS components)
    add_dependencies(${component} ffmpeg_build)
endforeach()
set_property(TARGET swresample PROPERTY INTERFACE_LINK_LIBRARIES avutil)
set_property(TARGET avcodec PROPERTY INTERFACE_LINK_LIBRARIES "avutil;swresample")
set_property(TARGET avformat PROPERTY INTERFACE_LINK_LIBRARIES "avcodec;avutil")
set_property(TARGET swscale PROPERTY INTERFACE_LINK_LIBRARIES avutil)
if(FFMPEG_BUILD_PROGRAMS)
    set_property(TARGET avfilter PROPERTY INTERFACE_LINK_LIBRARIES "avformat;avcodec;swscale;swresample;avutil")
    set_property(TARGET avdevice PROPERTY INTERFACE_LINK_LIBRARIES "avfilter;avformat;avcodec;avutil")
endif()
message(STATUS "FFmpeg ${FFMPEG_VERSION}: ${profile} ${CMAKE_BUILD_TYPE}, ${CMAKE_SIZEOF_VOID_P}-byte pointers, install ${FFMPEG_INSTALL_DIR}")
